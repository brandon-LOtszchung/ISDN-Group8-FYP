🗄️ Database Schema 
%%{init: {'theme': 'base'}}%%

erDiagram
    families ||--o{ family_members : "has"
    families ||--o{ inventory_items : "owns"
    families ||--o{ saved_recipes : "saves"
    families ||--o{ shopping_list_items : "creates"

    families {
        uuid id PK
        text name
        enum cooking_skill
        enum budget_range
        enum language
    }

    family_members {
        uuid id PK
        uuid family_id FK
        text name
        int age
        array dietary_restrictions
        array allergies
    }

    inventory_items {
        uuid id PK
        uuid family_id FK
        text name
        enum category
        numeric quantity
    }

    saved_recipes {
        uuid id PK
        uuid family_id FK
        jsonb recipe_data
        text cuisine_style
        numeric cost
    }

    shopping_list_items {
        uuid id PK
        uuid family_id FK
        text name
        numeric quantity
        boolean is_purchased
    }
