create table public.saved_recipes (
  id uuid not null default extensions.uuid_generate_v4 (),
  family_id uuid not null,
  recipe_data jsonb not null,
  is_favorite boolean null default false,
  created_at timestamp with time zone null default now(),
  cuisine_style text null,
  matched_count integer not null default 0,
  total_count integer not null default 0,
  estimated_total_cost numeric(10, 2) null,
  constraint saved_recipes_pkey primary key (id),
  constraint saved_recipes_family_id_fkey foreign KEY (family_id) references families (id) on delete CASCADE,
  constraint saved_recipes_counts_nonnegative check (
    (
      (matched_count >= 0)
      and (total_count >= 0)
      and (matched_count <= total_count)
    )
  ),
  constraint saved_recipes_estimated_total_cost_nonnegative check (
    (
      (estimated_total_cost is null)
      or (estimated_total_cost >= (0)::numeric)
    )
  )
) TABLESPACE pg_default;

create index IF not exists idx_saved_recipes_family_created on public.saved_recipes using btree (family_id, created_at desc) TABLESPACE pg_default;

create index IF not exists idx_saved_recipes_family_cuisine on public.saved_recipes using btree (family_id, cuisine_style) TABLESPACE pg_default;

create index IF not exists idx_saved_recipes_family_id on public.saved_recipes using btree (family_id) TABLESPACE pg_default;
create table public.shopping_list_items (
  id uuid not null default extensions.uuid_generate_v4 (),
  family_id uuid not null,
  name text not null,
  quantity numeric(10, 2) not null,
  unit text not null,
  alternatives text[] null default '{}'::text[],
  is_purchased boolean null default false,
  created_at timestamp with time zone null default now(),
  estimated_unit_cost numeric(10, 2) null,
  constraint shopping_list_items_pkey primary key (id),
  constraint shopping_list_items_family_id_name_unit_key unique (family_id, name, unit),
  constraint shopping_list_items_family_id_fkey foreign KEY (family_id) references families (id) on delete CASCADE,
  constraint shopping_list_items_estimated_unit_cost_nonnegative check (
    (
      (estimated_unit_cost is null)
      or (estimated_unit_cost >= (0)::numeric)
    )
  )
) TABLESPACE pg_default;

create index IF not exists idx_shopping_list_family_id on public.shopping_list_items using btree (family_id) TABLESPACE pg_default;

create table public.inventory_items (
  id uuid not null default extensions.uuid_generate_v4 (),
  family_id uuid not null,
  name text not null,
  category text not null,
  quantity numeric(10, 2) not null,
  expiry_date date null,
  added_at timestamp with time zone null default now(),
  updated_at timestamp with time zone null default now(),
  constraint inventory_items_pkey primary key (id),
  constraint inventory_items_family_id_fkey foreign KEY (family_id) references families (id) on delete CASCADE,
  constraint inventory_items_category_check check (
    (
      category = any (
        array[
          'vegetables'::text,
          'fruits'::text,
          'meat'::text,
          'seafood'::text,
          'dairy'::text,
          'grains'::text,
          'condiments'::text,
          'beverages'::text,
          'snacks'::text,
          'frozen'::text,
          'canned'::text,
          'other'::text
        ]
      )
    )
  )
) TABLESPACE pg_default;

create index IF not exists idx_inventory_family_id on public.inventory_items using btree (family_id) TABLESPACE pg_default;

create index IF not exists idx_inventory_category on public.inventory_items using btree (category) TABLESPACE pg_default;

create index IF not exists idx_inventory_expiry on public.inventory_items using btree (expiry_date) TABLESPACE pg_default;

create trigger update_inventory_items_updated_at BEFORE
update on inventory_items for EACH row
execute FUNCTION update_updated_at_column ();

create table public.family_members (
  id uuid not null default extensions.uuid_generate_v4 (),
  family_id uuid not null,
  name text not null,
  age integer not null,
  dietary_restrictions text[] null default '{}'::text[],
  allergies text[] null default '{}'::text[],
  created_at timestamp with time zone null default now(),
  updated_at timestamp with time zone null default now(),
  constraint family_members_pkey primary key (id),
  constraint family_members_family_id_fkey foreign KEY (family_id) references families (id) on delete CASCADE,
  constraint family_members_age_check check (
    (
      (age >= 1)
      and (age <= 120)
    )
  )
) TABLESPACE pg_default;

create index IF not exists idx_family_members_family_id on public.family_members using btree (family_id) TABLESPACE pg_default;

create trigger update_family_members_updated_at BEFORE
update on family_members for EACH row
execute FUNCTION update_updated_at_column ();

create table public.families (
  id uuid not null default extensions.uuid_generate_v4 (),
  name text not null,
  cooking_skill_level text not null,
  budget_range text not null,
  preferred_language text not null,
  created_at timestamp with time zone null default now(),
  updated_at timestamp with time zone null default now(),
  constraint families_pkey primary key (id),
  constraint families_budget_range_check check (
    (
      budget_range = any (array['low'::text, 'medium'::text, 'high'::text])
    )
  ),
  constraint families_cooking_skill_level_check check (
    (
      cooking_skill_level = any (
        array[
          'beginner'::text,
          'intermediate'::text,
          'advanced'::text
        ]
      )
    )
  ),
  constraint families_preferred_language_check check (
    (
      preferred_language = any (
        array[
          'en'::text,
          'zh-HK'::text,
          'fil'::text,
          'id'::text
        ]
      )
    )
  )
) TABLESPACE pg_default;

create trigger update_families_updated_at BEFORE
update on families for EACH row
execute FUNCTION update_updated_at_column ();
